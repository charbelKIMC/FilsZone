<?php
    $host = "localhost";
    $dbname = "wifi_zone";
    $username = "root";
    $password = "";

    try {
        $pdo = new PDO("mysql:host=$host;dbname=$dbname;charset=utf8", $username, $password);
        $pdo->setAttribute(PDO::ATTR_ERRMODE, PDO::ERRMODE_EXCEPTION);
    }catch(PDOException $e){
        die('Erreur Base de donnée : ' .$e->getMessage());
    }

    if(isset($_POST['ajouter'])) {
        $ajouter = $pdo->prepare("INSERT INTO forfait (nom, prix, volume_go, duree_validite, id_admin) VALUES (?,?,?,?,'1')");
        $ajouter->execute([$_POST['nom'], $_POST['prix'], $_POST['volume_go'], $_POST['duree_validite']]);
        header("Location: dashboard.php");
    }

    if(isset($_GET['supprimer'])){
        $supprimer = $pdo->prepare('DELETE FROM forfait WHERE id_forfait = ?');
        $supprimer -> execute([$_GET['supprimer']]);
        header('Location: dashboard.php');
    }

    if(isset($_POST['modifier'])) {
        $duree_validite = $_POST['duree_validite'];
        $volume_go = $_POST['volume_go'];
        if($duree_validite === '') {
            $duree_validite = null;
        }

        if($volume_go === ''){
            $volume_go = null;
        }
       
        $modifier = $pdo->prepare('UPDATE forfait SET nom = ?, prix = ?, volume_go = ?, duree_validite = ?  WHERE id_forfait = ?');
        $modifier -> execute([$_POST['nom'], $_POST['prix'], $volume_go, $duree_validite, $_POST['id_forfait']]);
        header('Location: dashboard.php');
    }

    $forfait = $pdo->query('SELECT * FROM forfait ORDER BY id_forfait DESC')->fetchALL();
    $editUser = null;
    if(isset($_GET['edit'])) {
        $edit = $pdo->prepare('SELECT * FROM forfait WHERE id_forfait = ?');
        $edit -> execute([$_GET['edit']]);
        $editUser = $edit->fetch();
    }
?>

<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>CRUD de wifi zone</title>
    <link href="https://cdn.jsdelivr.net/npm/bootstrap@5.3.0/dist/css/bootstrap.min.css" rel="stylesheet">
    <style>
        *{
            background: white;
            padding: 4px;
        }
        
    </style>
</head>
<body>
    <div class="container">
        <h3><?= $editUser ? "Modifier" : "Ajouter un forfait" ?></h3>
        <form action="" method="POST">
            <?php if($editUser): ?>
                <input type="hidden" name="id_forfait" value="<?= $editUser['id_forfait'] ?>">
            <?php endif; ?>
            <div>
                <label for="nom">Nom</label>
                <input type="text" name="nom" required value="<?= $editUser['nom'] ?? '' ?>">
            </div>
            <div>
                <label for="prix">Prix</label>
                <input type="text" name="prix"  value="<?= $editUser['prix'] ?? '' ?>">
            </div>
            <div>
                <label for="volume">Volume de Go</label>
                <input type="text" name="volume_go"  value="<?= $editUser['volume_go'] ?? '' ?>">
            </div>
            <div>
                <label for="duree">Durée de validité</label>
                <input type="text" name="duree_validite" value="<?= $editUser['duree_validite'] ?? '' ?>">
            </div>
            <div>
                <button name="<?= $editUser ? 'modifier' : 'ajouter' ?>"><?= $editUser ? 'Modifier' : 'Ajouter' ?></button>
            </div>
        </form>

        <h4>Liste (<?= count($forfait) ?>)</h4>
        <table>
            <tr>
                <th>ID</th>
                <th>Nom</th>
                <th>Prix</th>
                <th>Volume</th>
                <th>Durée de validité</th>
                <th>Action</th>
            </tr>
            <?php foreach($forfait as $f): ?>
                <tr>
                    <td><?= $f['id_forfait'] ?></td>
                    <td><?= htmlspecialchars($f['nom']) ?></td>
                    <td><?= htmlspecialchars($f['prix']) ?></td>
                    <td><?= htmlspecialchars($f['volume_go']) ?></td>
                    <td><?= htmlspecialchars($f['duree_validite']) ?></td>
                    <td>
                        <a href="?edit=<?= $f['id_forfait'] ?>">Edit</a>
                        <a href="?supprimer=<?= $f['id_forfait'] ?>" onclick="return confirm('Supprimer ?')">Supprimer</a>
                    </td>
                </tr>
            <?php endforeach;?>
        </table>
    </div>
</body>
</html>
